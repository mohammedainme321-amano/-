تشخيص الاتصال بـ Supabase + كود إنشاء الجداول (انسخه مرة واحدة)
  • جارٍ الفحص...
  1. افتح مشروعك في Supabase ثم SQL Editor ثم New query.
  2. اضغط «نسخ كود SQL» والصقه ثم Run (يمكن تشغيله أكثر من مرة بأمان).
  3. لإنشاء مستخدم للإضافة والتعديل: Authentication ← Users ← Add user (فعّل Auto Confirm).
  4. ارجع هنا واضغط «تحديث».
create table if not exists monthly_plans (id bigint generated always as identity primary key);
alter table monthly_plans
  add column if not exists title text,
  add column if not exists department text,
  add column if not exists assignee text,
  add column if not exists start_date date,
  add column if not exists end_date date,
  add column if not exists deliverables text,
  add column if not exists status text default 'معتمدة';

create table if not exists tasks (id bigint generated always as identity primary key);
alter table tasks
  add column if not exists title text,
  add column if not exists department text,
  add column if not exists assignee text,
  add column if not exists status text default 'جارية',
  add column if not exists progress int not null default 0,
  add column if not exists stumbling_reason text;

create table if not exists challenges (id bigint generated always as identity primary key);
alter table challenges
  add column if not exists title text,
  add column if not exists department text,
  add column if not exists severity text default 'متوسط',
  add column if not exists status text default 'مفتوح';

create table if not exists pending_decisions (id bigint generated always as identity primary key);
alter table pending_decisions
  add column if not exists title text,
  add column if not exists owner text,
  add column if not exists status text default 'معلق';

grant select on monthly_plans, tasks, challenges, pending_decisions to anon, authenticated;
grant insert, update, delete on monthly_plans, tasks, challenges, pending_decisions to authenticated;

do $$
declare t text;
begin
  foreach t in array array['monthly_plans','tasks','challenges','pending_decisions'] loop
    execute format('alter table %I enable row level security', t);
    execute format('drop policy if exists public_read on %I', t);
    execute format('drop policy if exists auth_insert on %I', t);
    execute format('drop policy if exists auth_update on %I', t);
    execute format('drop policy if exists auth_delete on %I', t);
    execute format('create policy public_read on %I for select to anon, authenticated using (true)', t);
    execute format('create policy auth_insert on %I for insert to authenticated with check (true)', t);
    execute format('create policy auth_update on %I for update to authenticated using (true) with check (true)', t);
    execute format('create policy auth_delete on %I for delete to authenticated using (true)', t);
    begin
      execute format('alter publication supabase_realtime add table %I', t);
    exception when others then null;
    end;
  end loop;
end $$;

notify pgrst, 'reload schema';
